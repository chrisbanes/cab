import {
  chromium,
  webkit,
} from "/Users/chris/.t3/worktrees/ensemble/t3-a3b2ba8d/node_modules/playwright/index.mjs";
import { boundWorld } from "/Users/chris/.t3/worktrees/ensemble/t3-a3b2ba8d/dist/test/execution/bound-turn-recovery-fixture.js";
import { seedOperatorRecovery } from "/Users/chris/.t3/worktrees/ensemble/t3-a3b2ba8d/dist/test/fixtures/operator-web.js";
import { randomUUID } from "node:crypto";
import { mkdir, writeFile } from "node:fs/promises";
const dir = "/tmp/cab-project-drain-20261103/1419-parity-work/captures-packet";
await mkdir(dir, { recursive: true });
const cleanup = [];
const t = {
  after(fn) {
    cleanup.push(fn);
  },
};
let browsers = [];
try {
  const w = await boundWorld(t);
  const f = w.s.f,
    ids = w.g,
    d = f.service.domain();
  seedOperatorRecovery(f, {
    projectId: ids.projectId,
    taskId: ids.taskId,
    assignmentId: ids.assignmentId,
  });
  f.seedPersistedState((db) =>
    db
      .prepare("INSERT INTO task_archival_holds (taskId,reason) VALUES (?,?)")
      .run(ids.taskId, "Review held work"),
  );
  d.execute({
    type: "project.configure",
    actor: "operator",
    key: randomUUID(),
    projectId: ids.projectId,
    expectedVersion: Number(d.project(ids.projectId).version),
    instructions: "Updated fixture instructions",
  });
  const web = await f.startWeb();
  const manifest = {
    candidate: process.env.CAB_PARITY_HEAD,
    tree: process.env.CAB_PARITY_TREE,
    fixture:
      "boundWorld codex fake runtime, real SQLite, 21 synthetic recovery records plus bound turn, stale project instructions, task hold",
    engines: [],
    screens: [],
    observations: [],
  };
  for (const [engine, kind] of [
    ["chromium", chromium],
    ["webkit", webkit],
  ]) {
    const browser = await kind.launch();
    browsers.push(browser);
    manifest.engines.push({ name: engine, version: browser.version() });
    const page = await browser.newPage({
      viewport: { width: 1366, height: 900 },
    });
    page.setDefaultTimeout(10000);
    await page.goto(
      web.origin + `/app/assignments/${ids.assignmentId}/recovery`,
    );
    await page.getByLabel("Password").fill(web.password);
    await page.getByRole("button", { name: "Sign in", exact: true }).click();
    await page
      .getByRole("heading", { name: "Execution and assignment" })
      .waitFor();
    const get = async (path) => {
      const response = await page.request.get(web.origin + path);
      if (response.status() !== 200)
        throw Error(`${path} ${response.status()}`);
      return response.text();
    };
    const runtime = await get(`/runtime/assignment/${ids.assignmentId}`),
      legacy = await get(`/assignment/${ids.assignmentId}`);
    manifest.observations.push({
      engine,
      runtimeStatus: 200,
      assignmentStatus: 200,
      runtimeForms: (runtime.match(/<form\b/g) || []).length,
      runtimeApply: runtime.includes("/runtime/control/instruction-apply"),
      oldRecords: (runtime.match(/<h3>Recovery record:/g) || []).length,
      omission: runtime.includes("older records are omitted"),
      oldLifecycle: legacy.includes("does not confirm runtime admission"),
      legacyRuntimeLink: legacy.includes("/runtime/assignment/"),
    });
    const mask = [
      page
        .locator(".settings-panel")
        .filter({
          has: page.getByRole("heading", { name: "Reconcile a bound turn" }),
        }),
      page.locator("details.assignment-facts > p"),
    ];
    async function shot(name, viewport) {
      await page.screenshot({
        path: `${dir}/${engine}-${name}.png`,
        mask,
        maskColor: "#37383b",
        animations: "disabled",
      });
      manifest.screens.push({ file: `${engine}-${name}.png`, viewport });
    }
    await shot("recovery-desktop", "1366x900");
    const link = page.getByRole("link", {
      name: "Inspect assignment to apply current instructions",
    });
    await link.focus();
    await page.keyboard.press(
      engine === "webkit" ? "Alt+Shift+Tab" : "Shift+Tab",
    );
    await page.keyboard.press(engine === "webkit" ? "Alt+Tab" : "Tab");
    if (!(await link.evaluate((e) => e === document.activeElement)))
      throw Error("keyboard focus not on link");
    await shot("keyboard-focus", "1366x900");
    await page.keyboard.press("Enter");
    const facts = page.locator(
      `[data-record-id="assignment:${ids.assignmentId}"] details.assignment-facts`,
    );
    await facts
      .getByRole("button", { name: /Apply current instructions/ })
      .waitFor();
    await facts
      .getByRole("button", { name: /Apply current instructions/ })
      .scrollIntoViewIfNeeded();
    await shot("inspector", "1366x900");
    await page.setViewportSize({ width: 390, height: 844 });
    await page.goto(
      web.origin + `/app/assignments/${ids.assignmentId}/recovery`,
    );
    await page
      .getByRole("heading", { name: "Execution and assignment" })
      .waitFor();
    await shot("recovery-phone", "390x844");
    await f.service.resumeTask(ids.taskId);
    await page.reload();
    await page.getByText("Execution held", { exact: true }).waitFor();
    await shot("held-phone", "390x844");
    await browser.close();
    browsers = browsers.filter((b) => b !== browser);
  }
  await writeFile(
    `${dir}/observations.json`,
    JSON.stringify(manifest, null, 2),
  );
  console.log(JSON.stringify(manifest));
} finally {
  for (const b of browsers) await b.close();
  for (const fn of cleanup.reverse()) await fn();
}
